"""Detector de secretos enfocado en credenciales de IA / gateway.

Cobertura específica de IA (keys de proveedores de modelos, endpoints, tokens del
gateway) que un secret-scanner genérico clasifica débilmente. Si gitleaks está
disponible en el runner, el .gitlab-ci.yml lo corre en paralelo; este detector es
el fallback regex y agrega contexto de dominio IA.
"""
from __future__ import annotations

import re
from pathlib import Path

from ..models import Finding, ScanResult, Severity, Standard

# (id, descripción, regex)
PATTERNS: list[tuple[str, str, re.Pattern]] = [
    ("openai", "OpenAI API key", re.compile(r"\bsk-[A-Za-z0-9]{20,}\b")),
    ("openai-proj", "OpenAI project key", re.compile(r"\bsk-proj-[A-Za-z0-9_-]{20,}\b")),
    ("anthropic", "Anthropic API key", re.compile(r"\bsk-ant-[A-Za-z0-9_-]{20,}\b")),
    ("hf", "HuggingFace token", re.compile(r"\bhf_[A-Za-z0-9]{30,}\b")),
    ("aws", "AWS access key", re.compile(r"\bAKIA[0-9A-Z]{16}\b")),
    ("gcp", "Google API key", re.compile(r"\bAIza[0-9A-Za-z_-]{35}\b")),
    ("cohere", "Cohere/genérica larga", re.compile(r"\bco-[A-Za-z0-9]{40}\b")),
    ("bearer", "Bearer token hardcodeado",
     re.compile(r"(?i)authorization\s*[:=]\s*['\"]?bearer\s+[A-Za-z0-9._-]{20,}")),
    ("generic-key", "Asignación de api_key/token literal",
     re.compile(r"(?i)(api[_-]?key|secret|token|password)\s*[:=]\s*['\"][^'\"\s]{12,}['\"]")),
    ("litellm-master", "LiteLLM master/salt key",
     re.compile(r"(?i)(LITELLM_MASTER_KEY|LITELLM_SALT_KEY)\s*[:=]\s*['\"]?sk-[^\s'\"]+")),
]

# Reduce ruido: placeholders evidentes.
PLACEHOLDER = re.compile(r"(?i)(your[_-]?key|xxx+|placeholder|example|dummy|<[^>]+>|\$\{)")

SRC_SUFFIXES = {".py", ".js", ".ts", ".yaml", ".yml", ".env", ".json", ".toml",
                ".ini", ".cfg", ".sh", ".tf", ".properties", ""}
SKIP_DIRS = {".git", ".venv", "venv", "node_modules", "__pycache__", "dist", "build"}


def detect(root: Path, result: ScanResult, changed_files: set[str] | None = None) -> None:
    for path in root.rglob("*"):
        if path.is_dir() or any(p in SKIP_DIRS for p in path.parts):
            continue
        if path.suffix.lower() not in SRC_SUFFIXES and path.name not in (".env",):
            continue
        try:
            lines = path.read_text(encoding="utf-8", errors="ignore").splitlines()
        except OSError:
            continue
        rel = path.relative_to(root).as_posix()
        if not result.should_scan(rel):
            continue
        for i, ln in enumerate(lines, 1):
            for sid, desc, pat in PATTERNS:
                m = pat.search(ln)
                if not m:
                    continue
                if PLACEHOLDER.search(m.group(0)):
                    continue
                # Las keys de proveedores de modelos son blocker; el resto high.
                sev = Severity.BLOCKER if sid in {
                    "openai", "openai-proj", "anthropic", "litellm-master"
                } else Severity.HIGH
                result.add(Finding(
                    check_id="SEC-001", domain="secrets", severity=sev,
                    title=f"Secreto hardcodeado: {desc}",
                    detail=f"Coincidencia en {rel}:{i} — '{m.group(0)[:12]}…' (redactado).",
                    file=rel, line=i,
                    remediation="Remover del repo, rotar la credencial y moverla a un "
                                "secret manager / variable CI protegida. Reescribir historia "
                                "si fue commiteada (git filter-repo).",
                    standards=[Standard.LEY_25326, Standard.BCRA, Standard.NIST_AI_RMF],
                    fingerprint=f"SEC-001:{sid}:{rel}:{i}",
                ))
